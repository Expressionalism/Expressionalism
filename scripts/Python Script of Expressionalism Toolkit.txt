def clamp(v: float, minv: float = 0.0, maxv: float = 1.0) -> float:
    return max(minv, min(maxv, float(v)))


def geometric_mean(values: Sequence[float]) -> float:
    vals = [float(x) for x in values if x is not None]
    if not vals:
        return 0.0
    product = 1.0
    for x in vals:
        if x <= 0.0:
            return 0.0
        product *= x
    return product ** (1.0 / len(vals))


def harmonic_mean(values: Sequence[float]) -> float:
    vals = [float(x) for x in values if x is not None and x > 0.0]
    if not vals:
        return 0.0
    return len(vals) / sum(1.0 / x for x in vals)


def linear_membership(x: float, tau: float = DEFAULT_TAU) -> float:
    """Formal Model §2 — linear (default) membership."""
    x, tau = float(x), float(tau)
    if x <= tau:
        return 0.0
    if x >= 1.0:
        return 1.0
    return clamp((x - tau) / (1.0 - tau))


def sigmoid_membership(
    x: float, tau: float = DEFAULT_TAU, k: float = 10.0, c: float = 0.55
) -> float:
    """Formal Model §2 — clamped sigmoid for high-tension / paradoxical inputs."""
    x, tau = float(x), float(tau)
    if x <= tau:
        return 0.0
    if x >= 1.0:
        return 1.0
    return clamp(1.0 / (1.0 + math.exp(-k * (x - c))))


def hybrid_H(mu: float, p_norm: float = 1.0) -> float:
    """Formal Model §1 — H(x) = min(1, μ(x) × P_norm(x | V))."""
    return min(1.0, float(mu) * float(p_norm))


def dilemma_gate(h: float, tau: float = DEFAULT_TAU) -> str:

def is_isolated(traction: float, tau: float = DEFAULT_TAU) -> bool:

def isolation_tag(full: bool = True) -> str:

def apply_hard_isolation(
    tractions: Sequence[float], tau: float = DEFAULT_TAU
) -> Tuple[List[int], List[int], float]:
    """Non-negotiable post-filter (Formal Model §6.2)."""
    non_iso: List[int] = []
    iso: List[int] = []
    for i, t in enumerate(tractions):
        (iso if is_isolated(t, tau) else non_iso).append(i)
    n = len(tractions)
    proportion = (len(iso) / n) if n > 0 else 0.0
    return non_iso, iso, proportion


def uncertainty_boost(
    base_u: float, proportion_isolated: float, coefficient: float = 0.05
) -> float:
    """Parameter Appendix §4.4 — +0.05 × proportion, clamped [0.25, 0.9]."""
    return clamp(float(base_u) + coefficient * float(proportion_isolated), 0.25, 0.9)


def secondary_coherence_score(
    alignments: Sequence[float],
    inter_coherences: Sequence[float],
    tractions: Sequence[float],
    tau: float = DEFAULT_TAU,
) -> Union[float, str]:
    """
    PRIMARY DIAGNOSTIC.
    SCS = 0.4·align + 0.35·inter + 0.25·traction
    Averages computed ONLY on traction > τ.
    """
    filtered = [
        (a, i, t)
        for a, i, t in zip(alignments, inter_coherences, tractions)
        if float(t) > float(tau)
    ]
    if not filtered:
        return "N/A (all components isolated)"
    aligns, inters, tracts = zip(*filtered)
    scs = (
        0.4 * float(np.mean(aligns))
        + 0.35 * float(np.mean(inters))
        + 0.25 * float(np.mean(tracts))
    )
    return clamp(scs, 0.0, 1.0)


def compute_relator_mode(
    R: float, D: float, T_norm: float, RT_norm: float, U_norm: float, Y_centered: float
) -> Dict[str, float]:
    """
    Formal Model §6.3 / Parameter Appendix §4.3.
    Softmax over the two relational modes only; Residual = complement.
    """
    R, D, T_norm, RT_norm, U_norm = map(clamp, (R, D, T_norm, RT_norm, U_norm))
    Y_centered = float(Y_centered)
    S_logit = 1.2 * R + 1.0 * D - 0.8 * T_norm
    O_logit = 1.5 * T_norm - 1.8 * RT_norm - 1.2 * U_norm + 0.9 * Y_centered
    max_logit = max(S_logit, O_logit)
    exp_S = math.exp(S_logit - max_logit)
    exp_O = math.exp(O_logit - max_logit)
    total = exp_S + exp_O
    p_self = exp_S / total
    p_other = exp_O / total
    residual = max(0.0, 1.0 - (p_self + p_other))
    return {
        "P(Self-reflexive)": round(p_self, 6),
        "P(Pointing to something else)": round(p_other, 6),
        "Residual": round(residual, 6),
    }


def spt_harmonic(stability_scores: Sequence[float]) -> float:
    """Harmonic mean of presumption stability scores, clamped [0.3, 0.7]."""
    return clamp(harmonic_mean(stability_scores), 0.3, 0.7)


def presumption_tempered_yield(tensive: float, spt: float) -> float:
    """Presumption-Tempered = Tensive × (1 − 0.1 × SPT)"""
    return clamp(float(tensive) * (1.0 - 0.1 * float(spt)))


def ascii_bar(value: float, width: int = 20) -> str:
    if not isinstance(value, (int, float)):
        return "N/A"
    filled = int(round(clamp(value) * width))
    return "█" * filled + "░" * (width - filled)


def tetralemma_label(value: float) -> str:
    if value > 0.8:
        return "affirm"
    if value < 0.2:
        return "negate"
    if 0.45 < value < 0.55:
        return "both (paradox)"
    return "neither (pause)"

class SecondaryAlignmentLedger:
    """Applies hard isolation exactly once; exposes immutable non-isolated view."""

    def __init__(self, secondaries: Sequence[SecondaryExpression], tau: float = DEFAULT_TAU):
        self._all: List[SecondaryExpression] = list(secondaries)
        self.tau = float(tau)
        self._isolation: Optional[IsolationResult] = None
        self._apply_isolation()

    def _apply_isolation(self) -> None:
        tractions = [s.traction for s in self._all]
        non_idx, iso_idx, prop = apply_hard_isolation(tractions, self.tau)
        non_isolated: List[SecondaryExpression] = []
        isolated: List[SecondaryExpression] = []
        for i, sec in enumerate(self._all):
            if i in iso_idx:
                tagged = SecondaryExpression(
                    id=sec.id, source=sec.source, type=sec.type,
                    alignment_score=sec.alignment_score, inter_coherence=sec.inter_coherence,
                    traction=sec.traction, summary=sec.summary,
                    required_presumptions=list(sec.required_presumptions),
                    status=ISOLATION_LABEL_FULL,
                )
                isolated.append(tagged)
            else:
                st = (
                    "Strong Mesh" if sec.traction > 0.65
                    else "Partial Mesh" if sec.traction > 0.30
                    else "Low Traction"
                )
                clean = SecondaryExpression(
                    id=sec.id, source=sec.source, type=sec.type,
                    alignment_score=sec.alignment_score, inter_coherence=sec.inter_coherence,
                    traction=sec.traction, summary=sec.summary,
                    required_presumptions=list(sec.required_presumptions),
                    status=st,
                )
                non_isolated.append(clean)
        annotation = (
            f"Hard isolation applied — {len(isolated)} components excluded "
            f"as Synthesis Refusal remainders (proportion {prop:.1%})."
            if isolated else "No hard isolation required."
        )
        self._isolation = IsolationResult(
            non_isolated=non_isolated, isolated=isolated,
            proportion_isolated=prop, count_isolated=len(isolated),
            count_total=len(self._all), annotation=annotation,
        )

    @property
    def all(self) -> List[SecondaryExpression]:
        return list(self._all)

    @property
    def non_isolated(self) -> List[SecondaryExpression]:
        assert self._isolation is not None
        return list(self._isolation.non_isolated)

    @property
    def isolated(self) -> List[SecondaryExpression]:
        assert self._isolation is not None
        return list(self._isolation.isolated)

    @property
    def isolation(self) -> IsolationResult:
        assert self._isolation is not None
        return self._isolation

    @property
    def proportion_isolated(self) -> float:
        return self.isolation.proportion_isolated

    @property
    def all_isolated(self) -> bool:
        return self.isolation.all_isolated

    def compute_scs(self) -> Union[float, str]:
        ni = self.non_isolated
        if not ni:
            return "N/A (all components isolated)"
        return secondary_coherence_score(
            [s.alignment_score for s in ni],
            [s.inter_coherence for s in ni],
            [s.traction for s in ni],
            tau=self.tau,
        )

    def avg_traction_non_isolated(self) -> float:
        ni = self.non_isolated
        if not ni:
            return 0.0
        return sum(s.traction for s in ni) / len(ni)

def _equity_alphas(query: str, equity: bool, emptiness_first: bool = False) -> List[float]:
    if not equity:
        return [1.0] * 5
    q = query.lower()
    void_keywords = (
        "void", "nothing", "paradox", "emptiness", "non-relational",
        "silence", "marks in sand", "gap", "refusal", "absence",
    )
    boost = 0.92 if any(k in q for k in void_keywords) or emptiness_first else 0.38
    return [boost, 0.33, 0.39, 0.41, 0.37]


def generate_secondary_expressions(
    primary: str,
    config: Optional[ToolkitConfig] = None,
    n: Optional[int] = None,
    seed: Optional[int] = None,
) -> List[SecondaryExpression]:
    """Seeded, equity-aware simulated secondary abduction (calculator mode)."""
    if config is None:
        config = ToolkitConfig()
    if n is None:
        n = config.max_secondaries
    n = max(3, min(int(n), 30))

    rng = random.Random(seed)
    np_rng = np.random.default_rng(seed)

    alphas = _equity_alphas(primary, config.secondary_expression_equity, config.emptiness_first)
    priors = dirichlet.rvs(alphas, size=1, random_state=np_rng)[0]
    priors = priors / priors.sum()

    sources = np_rng.choice(SOURCES, size=n, p=priors)
    types_ = np_rng.choice(TYPES, size=n, p=[0.48, 0.32, 0.20])
    base_align = np.clip(np_rng.normal(0.71, 0.12, n), 0.32, 0.97)

    # Isolation rate higher under emptiness / void keywords
    zero_prob = 0.12
    if config.emptiness_first or any(
        k in primary.lower()
        for k in ("void", "nothing", "silence", "emptiness", "non-relational", "gap")
    ):
        zero_prob = 0.28

    secondaries: List[SecondaryExpression] = []
    for i in range(n):
        traction = 0.0 if rng.random() < zero_prob else round(rng.uniform(0.12, 0.93), 3)
        inter = round(rng.uniform(0.48, 0.89), 3)
        align = round(float(base_align[i]), 3)
        required = rng.sample(PRESUMPTIONS, k=rng.randint(2, 4))
        src, typ = str(sources[i]), str(types_[i])
        domain = (
            "scientific/relational" if src in ("web", "browse")
            else "lived/cultural" if src in ("x", "cultural")
            else "phenomenal"
        )
        summary = f"{src.upper()}: {primary[:55]}... ({domain})"
        secondaries.append(
            SecondaryExpression(
                id=i, source=src, type=typ, alignment_score=align,
                inter_coherence=inter, traction=traction, summary=summary,
                required_presumptions=required, status="Pending",
            )
        )

    if config.peircean_abduction:
        secondaries.sort(key=lambda s: s.alignment_score * s.inter_coherence, reverse=True)
    else:
        secondaries.sort(key=lambda s: s.alignment_score, reverse=True)

    # Re-assign sequential ids after sort
    return [
        SecondaryExpression(
            id=i, source=s.source, type=s.type, alignment_score=s.alignment_score,
            inter_coherence=s.inter_coherence, traction=s.traction, summary=s.summary,
            required_presumptions=list(s.required_presumptions), status=s.status,
        )
        for i, s in enumerate(secondaries)
    ]

def stage1_scan(primary: str, config: Optional[ToolkitConfig] = None) -> Stage1Result:
    if config is None:
        config = ToolkitConfig()
    lower = primary.lower()
    phenom_keys = (
        "feeling", "sensation", "gradient", "painting", "sky", "emotion",
        "marks in sand", "void", "silence", "non-verbal",
    )
    data_type = "phenomenon" if any(k in lower for k in phenom_keys) else "expression"
    proto_var = clamp(
        0.15 + 0.002 * len(primary) + (0.08 if data_type == "phenomenon" else 0),
        0.05, 0.55,
    )
    if config.emptiness_first or config.non_dual:
        proto_var = clamp(proto_var + 0.07, 0.05, 0.60)
    if config.alien_mode:
        proto_var = clamp(proto_var * 0.75, 0.05, 0.60)

    sigma = config.chaos_scale
    if data_type == "phenomenon":
        sigma = clamp(sigma + 0.06, 0.0, 0.5)
    if config.alien_mode:
        sigma = clamp(sigma + 0.10, 0.0, 0.5)

    base_u = clamp(0.28 + 0.35 * proto_var + 0.04 * sigma, 0.25, 0.70)
    Rel = 1 if proto_var > config.tau else 0
    return Stage1Result(
        data_type=data_type, proto_var=proto_var, sigma=sigma,
        base_uncertainty=base_u, Rel=Rel, input_str=primary,
    )


def build_certainties(
    non_isolated: Sequence[SecondaryExpression],
    rng: random.Random,
    max_items: int = 8,
) -> List[CertaintyItem]:
    """Phase 3 style — certainties derived only from non-isolated secondaries."""
    items: List[CertaintyItem] = []
    for s in list(non_isolated)[:max_items]:
        geo = clamp(s.alignment_score * 0.86, 0.27, 0.92)
        harm = clamp(geo * 0.93, 0.21, 0.87)
        items.append(
            CertaintyItem(
                id=s.id,
                summary=s.summary,
                yield_geo=round(geo, 3),
                yield_harm=round(harm, 3),
                alignment_score=s.alignment_score,
                inter_coherence=s.inter_coherence,
                traction=s.traction,
                probabilistic_relation=round(s.alignment_score, 3),
            )
        )
    return items


def build_uncertainties(
    isolated: Sequence[SecondaryExpression],
    non_isolated: Sequence[SecondaryExpression],
    u_global: float,
    rng: random.Random,
    max_items: int = 8,
) -> List[UncertaintyItem]:
    """Phase 4 — isolated always appear with exact full label; residual gaps added."""
    items: List[UncertaintyItem] = []
    # Isolated first (exact label required)
    for s in isolated:
        items.append(
            UncertaintyItem(
                id=s.id,
                incidence=int(13 + rng.randint(-7, 11)),
                prob=round(clamp(u_global + rng.uniform(-0.05, 0.08), 0.25, 0.85), 3),
                volume=round(rng.uniform(0.45, 0.92), 2),
                summary=f"Gap / isolated secondary {s.id}: {s.summary[:50]}...",
                linked_presumption=rng.choice(PRESUMPTIONS),
                traction=0.0,
                is_isolated=True,
                status_note=ISOLATION_LABEL_FULL,
            )
        )
    # Additional residual / low-traction style uncertainties from non-isolated
    remaining = max_items - len(items)
    for s in list(non_isolated)[: max(0, remaining)]:
        if s.traction < 0.40 or rng.random() < 0.35:
            items.append(
                UncertaintyItem(
                    id=s.id + 100,
                    incidence=int(8 + rng.randint(-4, 9)),
                    prob=round(clamp(u_global + rng.uniform(-0.04, 0.10), 0.25, 0.80), 3),
                    volume=round(rng.uniform(0.35, 0.78), 2),
                    summary=f"Residual gap linked to secondary {s.id}: {s.summary[:48]}...",
                    linked_presumption=rng.choice(PRESUMPTIONS),
                    traction=s.traction,
                    is_isolated=False,
                    status_note="Synthesis refused / low traction residual",
                )
            )
    return items[:max_items]


def build_tempered_and_presumptions(
    certainties: Sequence[CertaintyItem],
    u_global: float,
    config: ToolkitConfig,
    rng: random.Random,
) -> Tuple[List[TemperedCertainty], List[RequiredPresumption], float, float]:
    """Phase 5 — Reflexive Certainty Temper (only on non-isolated certainties)."""
    if not config.certainty_temper_probe or not certainties:
        return [], [], 0.0, 0.48

    tempered: List[TemperedCertainty] = []
    required: List[RequiredPresumption] = []
    doubt_scores: List[float] = []

    for cert in certainties[:7]:
        # Simulated layer / repeatability scores
        repeatability = clamp(0.35 + 0.45 * cert.alignment_score + rng.uniform(-0.08, 0.08), 0.20, 0.88)
        relator_score = clamp(0.30 + 0.50 * cert.inter_coherence + rng.uniform(-0.07, 0.07), 0.22, 0.85)
        layer_score = clamp((repeatability + relator_score) / 2 + rng.uniform(-0.05, 0.05), 0.18, 0.82)
        avg_layer = (repeatability + relator_score + layer_score) / 3
        tet = tetralemma_label(avg_layer)

        n_pres = rng.randint(2, 4)
        abducted = rng.sample(PRESUMPTIONS, k=n_pres)
        chosen = rng.choice(abducted)
        doubt_trigger = f"Unaddressed presumption: {chosen} (tetralemma: {tet})"
        spectrum_stability = clamp(avg_layer, 0.28, 0.72)
        doubt_scores.append(spectrum_stability)

        temper_index = clamp(1.0 - (layer_score * (1.0 - 0.13 * u_global)), 0.25, 0.72)
        tempered_geo = clamp(rng.uniform(0.16, 0.32), 0.15, 0.35)

        tempered.append(
            TemperedCertainty(
                original_summary=cert.summary[:70] + ("..." if len(cert.summary) > 70 else ""),
                doubt_trigger=doubt_trigger,
                tempered_yield_geo=round(tempered_geo, 3),
                tempered_yield_harm=round(clamp(tempered_geo * 0.83, 0.12, 0.30), 3),
                certainty_uncertainty_ratio=round(rng.uniform(0.32, 0.58), 3),
                summary=f"Tempers as {chosen.split(':')[0]} — lingers spectrum relativity.",
                required_presumption=chosen,
                certainty_temper_index=round(temper_index, 3),
                tetralemma=tet,
            )
        )

        for pres in abducted:
            required.append(
                RequiredPresumption(
                    presumption=pres,
                    linked_secondary_id=cert.id,
                    spectrum_stability=round(spectrum_stability, 3),
                    uncertainty_prob=round(
                        clamp(u_global + rng.uniform(-0.07, 0.07), 0.25, 0.78), 3
                    ),
                    shaky_flag=spectrum_stability < 0.50,
                    temper_note="Tensive / shaky / odd" if spectrum_stability < 0.50 else "Stable",
                )
            )

    spt = spt_harmonic(doubt_scores) if doubt_scores else 0.48
    avg_doubt = (
        float(np.mean([t.certainty_uncertainty_ratio for t in tempered]))
        if tempered else 0.0
    )
    return tempered, required, avg_doubt, spt

def compute_metrics(
    ledger: SecondaryAlignmentLedger,
    config: ToolkitConfig,
    stage1: Stage1Result,
    seed: Optional[int] = None,
) -> MetricsBundle:
    """Full MetricsBundle including operational tables, after isolation."""
    rng = random.Random(seed)
    isolation = ledger.isolation
    all_iso = isolation.all_isolated
    prop = isolation.proportion_isolated

    u_global = uncertainty_boost(stage1.base_uncertainty, prop)
    scs = ledger.compute_scs()

    ni = ledger.non_isolated
    if ni:
        aligns = [s.alignment_score for s in ni]
        mean_align = float(np.mean(aligns))
        resonance = clamp(1.0 - float(np.mean([abs(a - mean_align) for a in aligns])), 0.25, 0.95)
        stability = clamp(1.0 - float(np.std(aligns)), 0.35, 0.97)
        avg_traction = float(np.mean([s.traction for s in ni]))
        stab_scores = [
            clamp(0.4 * s.alignment_score + 0.6 * s.inter_coherence, 0.15, 0.95)
            for s in ni
        ]
        spt_base = spt_harmonic(stab_scores)
    else:
        resonance = stability = avg_traction = 0.0
        spt_base = 0.30

    # Operational generation (isolation-respecting)
    certainties = build_certainties(ni, rng)
    uncertainties = build_uncertainties(ledger.isolated, ni, u_global, rng)
    tempered, required, avg_doubt, spt = build_tempered_and_presumptions(
        certainties, u_global, config, rng
    )
    # Prefer Phase-5 SPT when available
    if tempered:
        spt = spt
    else:
        spt = spt_base
        avg_doubt = clamp(0.25 + 0.35 * prop + 0.15 * (1.0 - stability), 0.15, 0.70)

    uncertainty_resonance = clamp(1.0 - 0.35 * prop - (0.15 if all_iso else 0.0), 0.20, 0.80)
    uncertainty_stability = clamp(0.55 + 0.10 * (1.0 - prop), 0.40, 0.85)

    reflexive_tension = clamp(avg_doubt / (0.5 + resonance + 0.01), 0.20, 0.70)

    # Density probe (optional)
    avg_density = 0.5
    layer_spectrum_avg = 0.5
    if config.density_probe and ni:
        avg_density = clamp(float(np.mean([s.inter_coherence for s in ni])), 0.22, 0.78)
        layer_spectrum_avg = clamp(float(np.mean([s.alignment_score for s in ni])), 0.18, 0.82)

    if ni and not all_iso:
        truth_alignment = clamp(0.82 * (1 - 0.12 * spt) * (1 - 0.11 * u_global), 0.25, 0.94)
        certainty_geo = geometric_mean([c.yield_geo for c in certainties]) if certainties else 0.5
        uncertainty_geo = clamp(0.30 + 0.25 * prop, 0.20, 0.70)
        meaning_tension = clamp(
            (certainty_geo / (certainty_geo + uncertainty_geo + 0.007))
            * (1.09 if spt < 0.5 else 1.0),
            0.20, 0.65,
        )
        metrics_for_yield = [
            resonance, stability, uncertainty_resonance, uncertainty_stability,
            truth_alignment, meaning_tension,
        ]
        sum_yield = float(sum(metrics_for_yield))
        geo_yield = geometric_mean(metrics_for_yield)
        harm_yield = harmonic_mean(metrics_for_yield) * (1 + 0.18 * u_global)
        tensive_yield = geo_yield * meaning_tension
        pret_yield = presumption_tempered_yield(tensive_yield, spt)
        relational_index = clamp(
            harm_yield * (1 + 0.19 / (1 + u_global)) - 0.08 * avg_doubt, 0.20, 0.94
        )
        harmony_index = clamp(harmonic_mean(metrics_for_yield), 0.45, 0.95)
        if relational_index > 0.55:
            yield_type = "Balanced presumption-tempered"
        elif relational_index > 0.35:
            yield_type = "Tensive"
        else:
            yield_type = "Tempered/raw"
        if avg_doubt > 0.40:
            yield_type += " (reflexive)"
    else:
        truth_alignment = meaning_tension = 0.25
        sum_yield = geo_yield = harm_yield = tensive_yield = pret_yield = 0.0
        relational_index, harmony_index = 0.20, 0.45
        yield_type = "Raw/Regressed (all components isolated)"

    yields = {
        "sum": round(sum_yield, 4),
        "geometric": round(geo_yield, 4),
        "harmonic": round(harm_yield, 4),
        "tensive": round(tensive_yield, 4),
        "presumption_tempered": round(pret_yield, 4),
    }

    T_norm = avg_traction
    RT_norm = prop
    U_norm = clamp((u_global - 0.25) / 0.65, 0.0, 1.0)
    y_map = {
        "Balanced presumption-tempered": 0.40,
        "Tensive": 0.10,
        "Tempered/raw": -0.15,
        "Raw/Regressed (all components isolated)": -0.35,
    }
    Y_centered = y_map.get(yield_type.split(" (")[0], 0.0)

    if all_iso:
        relator = {
            "P(Self-reflexive)": 0.0,
            "P(Pointing to something else)": 0.0,
            "Residual": 1.0,
        }
    else:
        relator = compute_relator_mode(
            R=reflexive_tension, D=avg_doubt, T_norm=T_norm,
            RT_norm=RT_norm, U_norm=U_norm, Y_centered=Y_centered,
        )

    metric_evolution = build_metric_evolution(
        stage1, scs, resonance, stability, uncertainty_resonance,
        uncertainty_stability, u_global, spt, isolation,
        config.density_probe, avg_density, layer_spectrum_avg,
    )

    return MetricsBundle(
        scs=scs,
        uncertainty_prob_global=round(u_global, 4),
        spt=round(spt, 4),
        relator_mode=relator,
        yields=yields,
        relational_index=round(relational_index, 4),
        harmony_index=round(harmony_index, 4),
        resonance=round(resonance, 4),
        stability=round(stability, 4),
        uncertainty_resonance=round(uncertainty_resonance, 4),
        uncertainty_stability=round(uncertainty_stability, 4),
        truth_alignment=round(truth_alignment, 4),
        meaning_tension=round(meaning_tension, 4),
        avg_doubt=round(avg_doubt, 4),
        yield_type=yield_type,
        isolation=isolation,
        all_isolated=all_iso,
        certainties=certainties,
        uncertainties=uncertainties,
        tempered=tempered,
        required_presumptions=required,
        metric_evolution=metric_evolution,
        avg_density=avg_density,
        layer_spectrum_avg=layer_spectrum_avg,
    )

def _scs_str(scs: Union[float, str]) -> str:
    return scs if isinstance(scs, str) else f"{scs:.3f}"


def _scs_note(scs: Union[float, str], all_isolated: bool, prop: float) -> str:
    if all_isolated or isinstance(scs, str):
        return "N/A (all components isolated)"
    strength = (
        "strong secondary synthesis traction" if scs > 0.65
        else "tensive / partial traction / Synthesis Refusal active"
    )
    extra = f" (computed on non-isolated components only; {prop:.0%} isolated)" if prop > 0 else ""
    return f"{scs:.3f} ({strength}){extra}"


def render_light_output(
    primary: str, stage1: Stage1Result, ledger: SecondaryAlignmentLedger, metrics: MetricsBundle
) -> str:
    scs_txt = _scs_str(metrics.scs)
    iso = metrics.isolation
    shaky = []
    if metrics.spt < 0.5:
        shaky.append("Secondary Presumption Temper is tensive/shaky")
    if iso.count_isolated:
        shaky.append(f"{iso.count_isolated} secondary expression(s) triggered Synthesis Refusal")
    shaky_txt = "; ".join(shaky) if shaky else "core assumptions remain provisional"
    return (
        f'This expression (“{primary[:60]}{"…" if len(primary) > 60 else ""}”) '
        f"connects through clear relational paths while leaving open questions. "
        f"Secondary Coherence Score = {scs_txt}. {shaky_txt}. "
        f"Overall it forms a useful provisional snapshot — truth holds reasonably well "
        f"(alignment ~{metrics.truth_alignment:.2f}) but carries productive tension "
        f"(meaning tension ~{metrics.meaning_tension:.2f}). "
        f"Yield type: {metrics.yield_type}. Always open to refinement per P8."
    )

def run_full_analysis(
    primary: str,
    config: Optional[ToolkitConfig] = None,
    n: Optional[int] = None,
    seed: Optional[int] = None,
) -> str:
    """End-to-end: Stage1 → abduct → Ledger → full operational metrics → canonical report."""
    if config is None:
        config = ToolkitConfig()

    stage1 = stage1_scan(primary, config)
    secs = generate_secondary_expressions(
        primary, config=config, n=n or config.max_secondaries, seed=seed
    )
    ledger = SecondaryAlignmentLedger(secs, tau=config.tau)

    # Optional simple meta-loop (re-abduct once if requested)
    if config.meta_loop and seed is not None:
        secs2 = generate_secondary_expressions(
            primary, config=config, n=n or config.max_secondaries, seed=seed + 1
        )
        # Prefer higher average traction non-isolated set
        led2 = SecondaryAlignmentLedger(secs2, tau=config.tau)
        if led2.avg_traction_non_isolated() > ledger.avg_traction_non_isolated():
            ledger = led2

    metrics = compute_metrics(ledger, config, stage1, seed=seed)
    return render_canonical_report(primary, stage1, ledger, metrics, config)
